EXTERN PRINTI: EXTERN INPUTI
GLOBAL I: GLOBAL ENTRY

*ENTRY
	GOSUB INPUTI
	N=I
	GOSUB PUTPRIMES
	RETURN

*ISPRIME
    IF I<4 RESULT=1: RETURN
    J=2: M=I

  *L0101
    IF (I MOD J)=0 RESULT=0: RETURN
    IF J<M J= J+1: M= I/J: GOTO *L0101

    RESULT= 1: RETURN

*PUTPRIMES
    I=1

    IF N=0 RETURN
  *L0102
  
    GOSUB ISPRIME
    IF RESULT=0 GOTO *L0202
      GOSUB PRINTI
      N= N-1
  *L0202

    I= I+1
    IF 0<N GOTO *L0102

  RETURN
